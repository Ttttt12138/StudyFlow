"""Implement this port in B. Claims and final writes must be atomic in B's DB."""

from typing import Protocol

from studyflow_ai.models import (
    AuditEvent,
    CandidatePlan,
    CheckResult,
    FinancialSnapshot,
    Forecast,
    IngestResult,
    PlanningContext,
)


class BackendPort(Protocol):
    def read_state(self, student_id: str) -> FinancialSnapshot: ...
    def forecast(self, snapshot: FinancialSnapshot, now) -> Forecast: ...
    def planning_context(self, snapshot, forecast, risk, now) -> PlanningContext: ...
    def validate_plan(self, plan: CandidatePlan, context: PlanningContext) -> CandidatePlan: ...
    def accept_import(self, student_id: str, result: IngestResult) -> int: ...
    def get_check(self, key: str) -> CheckResult | None: ...
    def claim_check(self, key: str) -> bool: ...
    def release_check(self, key: str) -> None: ...
    def save_check(self, key: str, result: CheckResult) -> bool:
        """Atomically compare state_version, persist check + audit/outbox, release claim."""
        ...

    def audit(self, event: AuditEvent) -> None: ...
